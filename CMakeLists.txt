cmake_minimum_required(VERSION 3.25)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" NOTIFICATION_HUB_VERSION_RAW)
string(STRIP "${NOTIFICATION_HUB_VERSION_RAW}" NOTIFICATION_HUB_VERSION)
if (NOTIFICATION_HUB_VERSION VERSION_LESS "0.1.5" OR NOTIFICATION_HUB_VERSION VERSION_GREATER "0.1.5")
  message(FATAL_ERROR "VERSION must be exactly 0.1.5, got '${NOTIFICATION_HUB_VERSION}'")
endif()

project(notification_hub_vnext VERSION ${NOTIFICATION_HUB_VERSION} LANGUAGES CXX)

include(CTest)
enable_testing()

add_subdirectory(runtime)
